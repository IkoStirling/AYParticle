cmake_minimum_required(VERSION 3.25)
project(AYParticle VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
# Standalone MSVC/Ninja needs the same UTF-8 /showIncludes repair as the engine.
# Probe both localized and English compilers rather than assuming a UI language.
if(PROJECT_IS_TOP_LEVEL AND MSVC AND CMAKE_GENERATOR MATCHES "Ninja")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/particle-prefix.h" "#pragma once\n")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/particle-prefix.cpp" "#include \"particle-prefix.h\"\n")
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" /nologo /utf-8 /showIncludes /Zs
        "${CMAKE_CURRENT_BINARY_DIR}/particle-prefix.cpp"
        OUTPUT_VARIABLE _particle_includes ERROR_VARIABLE _particle_include_errors ENCODING UTF-8)
    string(APPEND _particle_includes "${_particle_include_errors}")
    if(_particle_includes MATCHES "注意: 包含文件:  ")
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
        set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
    elseif(_particle_includes MATCHES "Note: including file:  ")
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "Note: including file:  ")
        set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "Note: including file:  ")
    endif()
    add_compile_definitions(AYPARTICLE_STANDALONE_DEPS_PREFIX=1)
endif()
add_library(AYParticle STATIC src/Particle.cpp src/ParticleProperty.cpp src/ParticleGeometry.cpp src/EffectAsset.cpp)
add_library(AYParticle::Runtime ALIAS AYParticle)
target_compile_features(AYParticle PUBLIC cxx_std_20)
target_include_directories(AYParticle PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
if(MSVC)
    target_compile_options(AYParticle PRIVATE /W4 /utf-8)
else()
    target_compile_options(AYParticle PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
    add_executable(AYParticle_Tests unittest/ParticleTests.cpp)
    target_link_libraries(AYParticle_Tests PRIVATE AYParticle)
    if(MSVC)
        target_link_options(AYParticle_Tests PRIVATE /DEBUG:NONE /INCREMENTAL:NO)
    endif()
    add_test(NAME AYParticle_Tests COMMAND AYParticle_Tests)
endif()
add_executable(AYParticle_Demo demo/ParticleDemo.cpp)
target_link_libraries(AYParticle_Demo PRIVATE AYParticle)
if(WIN32 AND TARGET AYRenderer AND TARGET AYShader AND TARGET bgfx::bgfx)
    # AYShader is a build-tool dependency; no compiler or source paths in the GPU runtime.
    add_executable(AYParticle_ShaderBake EXCLUDE_FROM_ALL tools/GpuParticleShaderBake.cpp)
    target_link_libraries(AYParticle_ShaderBake PRIVATE AYShader)
    set(_ay_particle_shader_dir "${CMAKE_CURRENT_BINARY_DIR}/generated/$<CONFIG>/d3d11")
    file(GLOB_RECURSE _ay_particle_shader_includes CONFIGURE_DEPENDS
        "${AY_SHADER_BGFX_SRC_DIR}/*.sh" "${AY_SHADER_BGFX_COMMON_DIR}/*.sh")
    add_custom_command(OUTPUT "${_ay_particle_shader_dir}/GpuParticleBinaries.h"
        BYPRODUCTS "${_ay_particle_shader_dir}/simulation.bin" "${_ay_particle_shader_dir}/diagnostic.bin"
            "${_ay_particle_shader_dir}/vertex.bin" "${_ay_particle_shader_dir}/fragment.bin"
            "${_ay_particle_shader_dir}/manifest.json"
        COMMAND $<TARGET_FILE:AYParticle_ShaderBake>
            --shaderc "${AY_SHADER_SHADERC_PATH}" --bgfx-include "${AY_SHADER_BGFX_SRC_DIR}"
            --common-include "${AY_SHADER_BGFX_COMMON_DIR}" --output "${_ay_particle_shader_dir}"
        DEPENDS AYParticle_ShaderBake src/GpuParticleShaders.h src/GpuShaderBinary.h
            "${AY_SHADER_SHADERC_PATH}" ${_ay_particle_shader_includes}
        COMMENT "Compiling and embedding GPU particle shaders with AYShader" VERBATIM)
    add_library(AYParticleGpu STATIC src/GpuParticle.cpp)
    add_library(AYParticle::Gpu ALIAS AYParticleGpu)
    target_link_libraries(AYParticleGpu PUBLIC AYParticle PRIVATE bgfx::bgfx)
    target_sources(AYParticleGpu PRIVATE "${_ay_particle_shader_dir}/GpuParticleBinaries.h")
    target_include_directories(AYParticleGpu PRIVATE "${_ay_particle_shader_dir}")
    if(BUILD_TESTING)
        add_executable(AYParticle_ShaderBinaryTests unittest/ShaderBinaryTests.cpp)
        target_sources(AYParticle_ShaderBinaryTests PRIVATE "${_ay_particle_shader_dir}/GpuParticleBinaries.h")
        target_compile_features(AYParticle_ShaderBinaryTests PRIVATE cxx_std_20)
        target_include_directories(AYParticle_ShaderBinaryTests PRIVATE "${_ay_particle_shader_dir}")
        add_test(NAME AYParticle_ShaderBinaryTests COMMAND AYParticle_ShaderBinaryTests)
    endif()
    add_executable(AYParticle_GpuDemo EXCLUDE_FROM_ALL demo/GpuParticleDemo.cpp)
    target_link_libraries(AYParticle_GpuDemo PRIVATE AYParticleGpu bgfx::bgfx)
    if(MSVC)
        target_compile_options(AYParticleGpu PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
        target_compile_options(AYParticle_ShaderBake PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
        target_link_options(AYParticle_ShaderBake PRIVATE /DEBUG:NONE /INCREMENTAL:NO)
        target_link_options(AYParticle_GpuDemo PRIVATE /DEBUG:NONE /INCREMENTAL:NO)
    endif()
endif()

                target_link_options(AYParticle_AssetTests PRIVATE /DEBUG:NONE /INCREMENTAL:NO)
            endif()
            add_test(NAME AYParticle_AssetTests COMMAND AYParticle_AssetTests)
